cmake_minimum_required(VERSION 3.18)
project(gowild_face LANGUAGES CXX)
add_library(facecore STATIC face_core.cpp vendor/facedetectcnn.cpp
    vendor/facedetectcnn-model.cpp vendor/facedetectcnn-data.cpp)
target_compile_features(facecore PUBLIC cxx_std_11)
set_target_properties(facecore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(facecore PRIVATE -O3 -fvisibility=hidden -ffunction-sections -fdata-sections)
# No OpenMP: the detector must never fan out over the renderer's cores.
if(ANDROID)
    if(ANDROID_ABI STREQUAL "armeabi-v7a" OR ANDROID_ABI STREQUAL "arm64-v8a")
        target_compile_definitions(facecore PRIVATE _ENABLE_NEON)
    endif()
    add_library(localface SHARED face_jni.cpp)
    target_link_libraries(localface PRIVATE facecore)
    target_link_options(localface PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
else()
    add_executable(face_core_test ../../../../../tools/tests/face_core_test.cpp)
    target_include_directories(face_core_test PRIVATE .)
    target_link_libraries(face_core_test PRIVATE facecore)
    enable_testing()
    add_test(NAME face_core COMMAND face_core_test)
endif()
